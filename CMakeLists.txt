cmake_minimum_required(VERSION 3.18)
project(m_scraps VERSION 0.6.0 LANGUAGES CXX)
if(DEFINED ENV{MPICXX})
  set(MPI_CXX_COMPILER "$ENV{MPICXX}")
elseif(APPLE AND EXISTS "/opt/homebrew/bin/mpic++")
  set(MPI_CXX_COMPILER "/opt/homebrew/bin/mpic++")
endif()
if(MPI_CXX_COMPILER)
  get_filename_component(M_SCRAPS_MPI_BIN "${MPI_CXX_COMPILER}" DIRECTORY)
  set(MPIEXEC_EXECUTABLE "${M_SCRAPS_MPI_BIN}/mpirun" CACHE FILEPATH "Matching MPI launcher")
endif()
find_package(MPI REQUIRED COMPONENTS CXX)
find_package(Eigen3 REQUIRED NO_MODULE)
add_executable(m_scraps_native SCRAPs/main.cpp SCRAPs/Lattice.cpp SCRAPs/readInput.cpp)
target_compile_features(m_scraps_native PRIVATE cxx_std_17)
target_link_libraries(m_scraps_native PRIVATE MPI::MPI_CXX Eigen3::Eigen)
set_target_properties(m_scraps_native PROPERTIES OUTPUT_NAME scraps)
install(TARGETS m_scraps_native RUNTIME DESTINATION m_scraps/SCRAPs)
install(DIRECTORY tools/ DESTINATION m_scraps/tools
        FILES_MATCHING PATTERN "*.py" PATTERN "tests" EXCLUDE PATTERN "benchmark" EXCLUDE PATTERN "__pycache__" EXCLUDE)
install(DIRECTORY tools/fixtures/ DESTINATION m_scraps/tools/fixtures)
install(FILES docs/compare_b2_multisub.py DESTINATION m_scraps/docs)
install(FILES python/scraps_ase.py DESTINATION .)
